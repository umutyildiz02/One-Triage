cmake_minimum_required(VERSION 3.22.1)
project(vitaltriage_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS vt_compile=2)
set(CMAKE_JOB_POOL_COMPILE vt_compile)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Applies to our JNI and all child native targets; inspect packaged ELF before release.
add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
# Select optimized CPU code only after upstream runtime feature detection.
# Keep the portable CPU backend; never assume an x86_64 emulator has AVX2.
if(ANDROID_ABI STREQUAL "x86_64")
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/whisper.cpp whisper-build)
include(WhisperAbort.cmake)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/sentencepiece sentencepiece-build EXCLUDE_FROM_ALL)
target_link_libraries(sentencepiece-static PUBLIC log)
add_library(vt-sentencepiece SHARED sentencepiece-jni.cpp)
target_include_directories(vt-sentencepiece PRIVATE ../../../../third_party/sentencepiece/src)
target_link_libraries(vt-sentencepiece PRIVATE sentencepiece-static)
add_library(vt-whisper SHARED whisper-jni.cpp)
target_link_libraries(vt-whisper PRIVATE whisper android log)
if(ANDROID_ABI STREQUAL "x86_64")
    target_compile_definitions(vt-whisper PRIVATE VT_DYNAMIC_CPU)
endif()
